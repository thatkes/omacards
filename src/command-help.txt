# Help wording for the command paths in src/command-paths.txt.
# One block per path: Path, Purpose, Flags and Example on single lines.
# Blank lines separate blocks. bin/check-doc-paths reads this file as text;
# Cli::commandHelp() reads it from the embedded Qt resource.

Path: add-card
Purpose: Add a Card to a Deck.
Flags: -q Q, -a A, [-d D], [--type T], [--tags a,b], [--json], --help, -h
Example: omacards add-card -q "Two plus two?" -a four --json

Path: add-cards
Purpose: Add Cards from JSON or stdin, e.g. [{"question":"...","answer":"..."}].
Flags: --deck D, [--description TEXT], [--input FILE|-], [--json], --help, -h
Example: omacards add-cards --deck Calculus --input cards.json --json

Path: add-deck
Purpose: Create a Deck.
Flags: NAME, [--description TEXT], [--json], --help, -h
Example: omacards add-deck Calculus --json

Path: cards
Purpose: List Cards, optionally in one Deck.
Flags: [-d D], [--json], --help, -h
Example: omacards cards -d Calculus --json

Path: check
Purpose: Show decks, Cards, due count, math and Deck issues.
Flags: [--deck NAME], [--math], [--strict], [--json], --help, -h
Example: omacards check --deck Calculus --json

Path: config
Purpose: Show or change Settings.
Flags: [--new-card-allowance N], [--day-start-hour H], [--retention 0.70-0.99], [--json], --help, -h
Example: omacards config --new-card-allowance 7 --json

Path: cram
Purpose: Simulate a Cram session.
Flags: [-d D], [--steps N], [--json], --help, -h
Example: omacards cram -d Calculus --steps 4 --json

Path: decks
Purpose: List decks.
Flags: [--json], --help, -h
Example: omacards decks --json

Path: decks current
Purpose: Show the Current deck.
Flags: [--json], --help, -h
Example: omacards decks current --json

Path: delete-card
Purpose: Delete a Card.
Flags: -c ID, [-d D], [--json], --help, -h
Example: omacards delete-card -c cdbec825f20d --json

Path: delete-deck
Purpose: Delete a Deck.
Flags: NAME, [--json], --help, -h
Example: omacards delete-deck Calculus --json

Path: due
Purpose: List Cards due on the current Study day.
Flags: [-d D], [--json], --help, -h
Example: omacards due -d Calculus --json

Path: edit-card
Purpose: Edit a Card's Question, Answer, type or Tags.
Flags: -c ID, [-q Q], [-a A], [--type T], [--tags a,b], [--json], --help, -h
Example: omacards edit-card -c cdbec825f20d -a five --json

Path: edit-deck
Purpose: Change a Deck's description without adding Cards.
Flags: NAME, --description TEXT, [--json], --help, -h
Example: omacards edit-deck Calculus --description 'New description' --json

Path: import
Purpose: Import Cards from a file.
Flags: PATH, [-d D], [--json], --help, -h
Example: omacards import cards.json -d Calculus --json

Path: open
Purpose: Start the GUI.
Flags: --help, -h
Example: omacards open

Path: preview
Purpose: Preview a Deck's Cards. Math: supported $$...$$ and \(...\); $...$ stays literal. --format markdown returns unrendered source.
Flags: --deck D, [--card ID], [--format text|html|markdown], [--json], --help, -h
Example: omacards preview --deck Calculus --json

Path: rename-deck
Purpose: Rename a Deck.
Flags: OLD, NEW, [--json], --help, -h
Example: omacards rename-deck Calculus Analysis --json

Path: review
Purpose: Record a Rating for a Card.
Flags: -c ID, -r 1|3, [--json], --help, -h
Example: omacards review -c cdbec825f20d -r 3 --json

Path: search
Purpose: Search Cards.
Flags: TERM, [--deck D], [--json], --help, -h
Example: omacards search Taylor --json

Path: skill
Purpose: Manage the agent skill. Bare `omacards skill` shows status and help without installing.
Flags: [install], [--json], --help, -h
Example: omacards skill --help

Path: skill install
Purpose: Install the agent skill.
Flags: [--json], --help, -h
Example: omacards skill install --json

Path: skill status
Purpose: Report whether the agent skill is installed, where, and whether it matches the bundled source.
Flags: [--json], --help, -h
Example: omacards skill status --json

Path: stats
Purpose: Show study statistics.
Flags: [--json], --help, -h
Example: omacards stats --json

Path: use-deck
Purpose: Choose the Current deck.
Flags: NAME, [--json], --help, -h
Example: omacards use-deck Calculus --json

Path: warm
Purpose: Render a Deck's math SVGs ahead of study.
Flags: --deck D, [--json], --help, -h
Example: omacards warm --deck Calculus --json
